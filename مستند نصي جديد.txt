<iframe id="studyFrame" src="https://studyduel.lovable.app/chat"></iframe>

<script>
    const iframe = document.getElementById('studyFrame');
    
    iframe.addEventListener('load', function() {
        try {
            const iframeDoc = iframe.contentDocument || iframe.contentWindow.document;
            hideLovableInDocument(iframeDoc);
        } catch(e) {
            console.log('لا يمكن الوصول للإطار بسبب سياسة CORS');
        }
    });
    
    function hideLovableInDocument(doc) {
        if (!doc) return;
        
        const allElements = doc.querySelectorAll('*');
        allElements.forEach(el => {
            if (el.innerText && (el.innerText.includes('Lovable') || el.innerText.includes('lovable'))) {
                if (el.tagName !== 'BODY' && el.tagName !== 'HTML') {
                    el.style.display = 'none';
                }
            }
        });
        
        const footers = doc.querySelectorAll('footer, .footer, [class*="footer"]');
        footers.forEach(footer => {
            if (footer.innerText && footer.innerText.length < 500) {
                footer.style.display = 'none';
            }
        });
        
        const possibleBadges = doc.querySelectorAll('[class*="badge"], [class*="credit"], [class*="made"]');
        possibleBadges.forEach(badge => {
            badge.style.display = 'none';
        });
    }
    
    setInterval(() => {
        try {
            const iframeDoc = iframe.contentDocument || iframe.contentWindow.document;
            hideLovableInDocument(iframeDoc);
        } catch(e) {}
    }, 3000);
</script>
